%----------------------------------------------------------------------------------------
%	Shared layout for cv.tex (English) and cv-fr.tex (French)
%	Paper size and language are set by each file's \documentclass and babel line.

% ATS: write a ToUnicode map so ligatures and accents extract as plain letters.
% pdfTeX only; must come before the fonts are used.
\input{glyphtounicode}
\pdfgentounicode=1

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\usepackage[sfdefault,type1,scale=0.95]{inter}
\usepackage[protrusion=true,expansion=true,kerning=true]{microtype}
% Inter's Type 1 hyphen has wide side bearings ("France - KLM"); pull it in.
% This also breaks the -- ligature: type en dashes as the character itself.
\SetExtraKerning[unit=1em]{encoding=T1}{- = {-60,-60}}

\usepackage[hmargin=0.6in,top=0.5in,bottom=0.45in]{geometry}
\usepackage{xcolor}
\usepackage{enumitem}
\usepackage{titlesec}

\definecolor{accent}{HTML}{1F3A5F}
\definecolor{muted}{HTML}{4A4A4A}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}
\linespread{1.04}
\raggedright

% Section titles: plain text heading + rule, so parsers see a normal line.
\titleformat{\section}{\color{accent}\large\bfseries}{}{0pt}{}[\vspace{1.5pt}{\color{accent}\titlerule[0.6pt]}]
\titlespacing*{\section}{0pt}{6pt}{3.5pt}

\setlist[itemize]{label={\color{muted}\textbullet}, leftmargin=1.1em, labelsep=0.45em,
  itemsep=1.2pt, topsep=1.5pt, parsep=0pt, partopsep=0pt}

\newcommand{\sep}{\textcolor{muted}{\enspace|\enspace}}

% \entry{Title}{Organization, place}{Dates}
\newcommand{\entry}[3]{\par\textbf{#1}\sep#2\hfill\textcolor{muted}{#3}\par}

% \skill{Label}{items}; cv-fr.tex redefines \skillcolon for the French space before ":"
\newcommand{\skillcolon}{:}
\newcommand{\skill}[2]{\par\hangindent=1.1em\textbf{#1\skillcolon} #2\par}

\newcommand{\entrygap}{\vspace{4pt}}

\usepackage[hidelinks,pdfusetitle,unicode]{hyperref}
